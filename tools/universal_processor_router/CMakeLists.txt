cmake_minimum_required(VERSION 3.20)
project(UniversalProcessorRouter LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(universal_processor_router UniversalProcessorRouter.cpp)
target_compile_features(universal_processor_router PUBLIC cxx_std_20)
target_include_directories(universal_processor_router PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(universal_processor_router PUBLIC Threads::Threads)
if(WIN32)
    target_compile_definitions(universal_processor_router PRIVATE _WIN32_WINNT=0x0A00 NOMINMAX)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(universal_processor_router PRIVATE _GNU_SOURCE)
endif()
add_executable(universal_processor_router_demo main.cpp)
target_link_libraries(universal_processor_router_demo PRIVATE universal_processor_router)
